// Convert a title into a URL-friendly slug:
// lowercase, strip accents, and reduce any other run of characters to a
// single hyphen, with no leading or trailing hyphen.
export function slugify(input) {
  return input
    .normalize("NFKD")
    .replace(/\p{M}/gu, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
}
